   170→/** Minuscules + suppression des diacritiques pour la recherche plein texte. */
   171→function normalizeText(value: string): string {
   172→  return value
   173→    .toLowerCase()
   174→    .normalize('NFD')
   175→    .replace(/[\u0300-\u036f]/g, '')
   176→    .trim()
   177→}
   178→
   179→function questionHaystack(q: BankQuestion): string {
   180→  return normalizeText(
   181→    [
   182→      q.question,
   183→      q.expectedAnswer,
   184→      q.reasoning,
   185→      q.skill,
   186→      q.oralShort,
   187→      q.oralLong,
   188→      q.commonErrors.join(' '),
   189→    ].join(' '),
   190→  )
   191→}
   192→
   193→/**
   194→ * Tirage aléatoire biaisé : chaque question reçoit Math.random() + 1 si elle
   195→ * existe dans les statistiques avec correct < attempts (jamais entièrement
   196→ * réussie), ce qui la fait remonter dans le tri décroissant.
   197→ */
   198→function pickSessionQuestions(
   199→  pool: BankQuestion[],
   200→  stats: Record<string, QuestionStat>,
   201→): BankQuestion[] {
   202→  return pool
   203→    .map((question) => {
   204→      const stat = stats[question.id]
   205→      const bonus = stat && stat.correct < stat.attempts ? 1 : 0
   206→      return { question, key: Math.random() + bonus }
   207→    })
   208→    .sort((a, b) => b.key - a.key)
   209→    .slice(0, SESSION_SIZE)
   210→    .map((scored) => scored.question)
   211→}

   219→const MINUTE_MS = 60_000
   220→
   221→function subscribeToClock(onStoreChange: () => void) {
   222→  const timer = setInterval(onStoreChange, MINUTE_MS)
   223→  return () => clearInterval(timer)
   224→}
   225→
   226→function getClockSnapshot(): number {
   227→  return Math.floor(Date.now() / MINUTE_MS) * MINUTE_MS
   228→}
   229→
   230→function getServerClockSnapshot(): number {
   231→  return 0
   232→}
   233→
   234→function useNow(): number {
   235→  return useSyncExternalStore(subscribeToClock, getClockSnapshot, getServerClockSnapshot)
   236→}
