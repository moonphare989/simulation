   171→   171→function normalizeText(value: string): string {
   172→   172→  return value
   173→   173→    .toLowerCase()
   174→   174→    .normalize('NFD')
   175→   175→    .replace(/[\u0300-\u036f]/g, '')
   176→   176→    .trim()
   177→   177→}
   178→   178→
   179→   179→function questionHaystack(q: BankQuestion): string {
   180→   180→  return normalizeText(
   181→   181→    [
   182→   182→      q.question,
   183→   183→      q.expectedAnswer,
   184→   184→      q.reasoning,
   185→   185→      q.skill,
   186→   186→      q.oralShort,
   187→   187→      q.oralLong,
   188→   188→      q.commonErrors.join(' '),
   189→   189→    ].join(' '),
   190→   190→  )
   191→   191→}
   192→   192→
   193→   193→/**
   194→   194→ * Tirage aléatoire biaisé : chaque question reçoit Math.random() + 1 si elle
   195→   195→ * existe dans les statistiques avec correct < attempts (jamais entièrement
   196→   196→ * réussie), ce qui la fait remonter dans le tri décroissant.
   197→   197→ */
   198→   198→function pickSessionQuestions(
   199→   199→  pool: BankQuestion[],
   200→   200→  stats: Record<string, QuestionStat>,
   201→   201→): BankQuestion[] {
   202→   202→  return pool
   203→   203→    .map((question) => {
   204→   204→      const stat = stats[question.id]
   205→   205→      const bonus = stat && stat.correct < stat.attempts ? 1 : 0
   206→   206→      return { question, key: Math.random() + bonus }
   207→   207→    })
   208→   208→    .sort((a, b) => b.key - a.key)
   209→   209→    .slice(0, SESSION_SIZE)
   210→   210→    .map((scored) => scored.question)
   211→   211→}

   213→   213→// ---------------------------------------------------------------------
   214→   214→// Horloge (notions dues) : instantané aligné à la minute pour rester
   215→   215→// stable entre deux rendus ; instantané serveur à 0 pour éviter tout
   216→   216→// écart d’hydratation (aucune notion due avant le montage client).
   217→   217→// ---------------------------------------------------------------------
   218→   218→
   219→   219→const MINUTE_MS = 60_000
   220→   220→
   221→   221→function subscribeToClock(onStoreChange: () => void) {
   222→   222→  const timer = setInterval(onStoreChange, MINUTE_MS)
   223→   223→  return () => clearInterval(timer)
   224→   224→}
   225→   225→
   226→   226→function getClockSnapshot(): number {
   227→   227→  return Math.floor(Date.now() / MINUTE_MS) * MINUTE_MS
   228→   228→}
   229→   229→
   230→   230→function getServerClockSnapshot(): number {
   231→   231→  return 0
   232→   232→}
   233→   233→
   234→   234→function useNow(): number {
   235→   235→  return useSyncExternalStore(subscribeToClock, getClockSnapshot, getServerClockSnapshot)
   236→   236→}
